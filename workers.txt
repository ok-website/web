    <script>
        // 回到顶部功能
        const backToTopButton = document.getElementById('backToTop');

        window.addEventListener('scroll', () => {
            backToTopButton.classList.toggle('visible', window.scrollY > 300);
        }, { passive: true });

        backToTopButton.addEventListener('click', () => {
            window.scrollTo({ top: 0, behavior: 'smooth' });
        });

        // 窄屏悬浮二维码：点击展开/收起
        const qrFloat = document.getElementById('qrFloat');

        document.getElementById('qrFloatToggle').addEventListener('click', (e) => {
            e.stopPropagation();
            qrFloat.classList.toggle('expanded');
        });

        // Lightbox 功能
        const lightbox = document.getElementById('lightbox');
        const lightboxImg = document.getElementById('lightbox-img');
        const lightboxCaption = document.querySelector('.lightbox-caption');
        const currentIndexElement = document.getElementById('current-index');
        const totalImagesElement = document.getElementById('total-images');
        const zoomInstruction = document.querySelector('.zoom-instruction');

        const images = Array.from(document.querySelectorAll('.lightbox-trigger'), (trigger) => ({
            src: trigger.src,
            alt: trigger.alt || ''
        }));

        let currentImageIndex = 0;

        // 更新灯箱内容
        function updateLightbox() {
            const image = images[currentImageIndex];
            lightboxImg.src = image.src;
            lightboxImg.classList.remove('zoomed');
            lightboxCaption.textContent = image.alt;
            currentIndexElement.textContent = currentImageIndex + 1;
            zoomInstruction.textContent = '点击图片缩放';
        }

        // 打开灯箱
        function openLightbox(index) {
            currentImageIndex = index;
            updateLightbox();
            lightbox.classList.add('active');
            document.body.style.overflow = 'hidden'; // 防止背景滚动
        }

        // 关闭灯箱
        function closeLightbox() {
            lightbox.classList.remove('active');
            document.body.style.overflow = ''; // 恢复背景滚动
        }

        // 上一张 / 下一张图片
        function stepImage(step) {
            currentImageIndex = (currentImageIndex + step + images.length) % images.length;
            updateLightbox();
        }

        // 切换图片缩放
        function toggleZoom() {
            const zoomed = lightboxImg.classList.toggle('zoomed');
            zoomInstruction.textContent = zoomed ? '点击退出缩放' : '点击图片缩放';
        }

        // 为所有触发器添加事件监听器
        document.querySelectorAll('.lightbox-trigger').forEach((trigger, index) => {
            trigger.addEventListener('click', () => openLightbox(index));
        });

        // 关闭灯箱
        document.querySelector('.close-lightbox').addEventListener('click', closeLightbox);

        // 点击背景关闭灯箱
        lightbox.addEventListener('click', (e) => {
            if (e.target === lightbox) {
                closeLightbox();
            }
        });

        // 导航按钮
        document.querySelector('.lightbox-prev').addEventListener('click', (e) => {
            e.stopPropagation();
            stepImage(-1);
        });

        document.querySelector('.lightbox-next').addEventListener('click', (e) => {
            e.stopPropagation();
            stepImage(1);
        });

        // 键盘操作：ESC 关闭，方向键切换
        document.addEventListener('keydown', (e) => {
            if (!lightbox.classList.contains('active')) return;
            if (e.key === 'Escape') {
                closeLightbox();
            } else if (e.key === 'ArrowLeft') {
                stepImage(-1);
            } else if (e.key === 'ArrowRight') {
                stepImage(1);
            }
        });

        // 点击图片缩放
        lightboxImg.addEventListener('click', (e) => {
            e.stopPropagation();
            toggleZoom();
        });

        totalImagesElement.textContent = images.length;
    </script>
